    // Exponential Backoff Fetch for robust API calls
    async function fetchWithRetry(url, options, maxRetries = 5) {
        const delays = [1000, 2000, 4000, 8000, 16000];
        for (let i = 0; i < maxRetries; i++) {
            try {
                const response = await fetch(url, options);
                if (!response.ok) {
                    throw new Error(`HTTP error! status: ${response.status}`);
                }
                return await response.json();
            } catch (e) {
                if (i === maxRetries - 1) throw e;
                await new Promise(resolve => setTimeout(resolve, delays[i]));
            }
        }
    }

            const gain = ctx.createGain();
            osc.connect(gain); gain.connect(ctx.destination);

            if(type === 'correct') { osc.frequency.setValueAtTime(600, ctx.currentTime); osc.frequency.setValueAtTime(800, ctx.currentTime + 0.1); gain.gain.setValueAtTime(0.1, ctx.currentTime); osc.start(); osc.stop(ctx.currentTime + 0.3); }
            if(type === 'wrong') { osc.frequency.setValueAtTime(300, ctx.currentTime); osc.frequency.exponentialRampToValueAtTime(150, ctx.currentTime + 0.3); gain.gain.setValueAtTime(0.1, ctx.currentTime); osc.start(); osc.stop(ctx.currentTime + 0.3); }

    // Voice playback function allowing language parameters
    function playText(text, lang = 'en-US') {
        if ('speechSynthesis' in window) {
            window.speechSynthesis.cancel(); // Stop any currently playing audio
            const msg = new SpeechSynthesisUtterance(text);
            msg.lang = lang;
            if (lang === 'en-US') msg.rate = 0.9; // Slightly slower for English vocab
            window.speechSynthesis.speak(msg);

    // --- AI Feature 1: Generate Custom Topic ---
    async function generateCustomVocab(topic) {
        loadingMsg.innerText = `Generating ${topic} Vocab...✨`;
        loadingOverlay.style.display = 'flex';
        
        const prompt = `Generate 5 advanced English vocabulary multiple-choice questions for university students related to the topic: "${topic}".
        Provide exactly 4 options per word in Traditional Chinese (zh-TW).
        Output MUST be a JSON array of objects. Do not use markdown wrappers.`;

        const payload = {
            contents: [{ parts: [{ text: prompt }] }],
            systemInstruction: { parts: [{ text: "You are an expert English teacher. Respond ONLY with valid JSON." }] },
            generationConfig: {
                responseMimeType: "application/json",
                responseSchema: {
                    type: "ARRAY",
                    items: {
                        type: "OBJECT",
                        properties: {
                            word: { type: "STRING" },
                            phonetic: { type: "STRING", description: "IPA format e.g. /wɜːrd/" },
                            options: { type: "ARRAY", items: { type: "STRING" }, description: "4 options in Traditional Chinese" },
                            correct: { type: "INTEGER", description: "Index of the correct option (0-3)" }
                        },
                        required: ["word", "phonetic", "options", "correct"]
                    }
                }
            }
        };

        try {
            const result = await fetchWithRetry(endpoint, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const text = result.candidates[0].content.parts[0].text;
            const newWords = JSON.parse(text);
            if (newWords && newWords.length >= 5) {
                words = newWords;
            }
        } catch (error) {
            console.error("AI Generation failed. Falling back to default.", error);
            // Keeps default words array if it fails
        } finally {
            loadingOverlay.style.display = 'none';
        }
    }

    function renderBoard() {
        pathContainer.innerHTML = '';
        for(let i = 0; i <= state.totalTiles; i++) {
            const tile = document.createElement('div');
            tile.className = `tile ${i === 0 ? 'tile-start' : (i === state.totalTiles ? 'tile-end' : '')}`;
            tile.id = `tile-${i}`;
            tile.innerHTML = i === 0 ? 'START' : (i === state.totalTiles ? 'END<br><span style="font-size:0.6rem">⭐</span>' : i);
            pathContainer.appendChild(tile);
        }

        const pPawn = document.createElement('div'); pPawn.className = 'pawn'; pPawn.id = 'player-pawn'; pPawn.innerText = state.playerMascot;
        const nPawn = document.createElement('div'); nPawn.className = 'pawn npc-pawn'; nPawn.id = 'npc-pawn'; nPawn.innerText = state.npcMascot;
        pathContainer.appendChild(pPawn); pathContainer.appendChild(nPawn);
        updatePawnPositions();
    }

    function updatePawnPositions() {
        setTimeout(() => {
            const pTile = document.getElementById(`tile-${state.playerPos}`);
            const nTile = document.getElementById(`tile-${state.npcPos}`);
            const pPawn = document.getElementById('player-pawn');
            const nPawn = document.getElementById('npc-pawn');

            if(pTile && pPawn) {
                const isEven = state.playerPos % 2 === 0;
                pPawn.style.left = `${pTile.offsetLeft + 10}px`;
                pPawn.style.marginTop = isEven ? '-55px' : '-35px';

                document.querySelectorAll('.tile').forEach(t => t.classList.remove('active'));
                if(state.playerPos > 0 && state.playerPos < state.totalTiles) pTile.classList.add('active');
                
                pathContainer.scrollTo({ left: pTile.offsetLeft - 120, behavior: 'smooth' });
            }
            if(nTile && nPawn) {
                 const isEvenN = state.npcPos % 2 === 0;
                 nPawn.style.left = `${nTile.offsetLeft + 15}px`;
                 nPawn.style.marginTop = isEvenN ? '-65px' : '-45px';
            }
        }, 100);
    }

    function rollDice() {
        if(!state.canRoll) return;
        
        sfx('roll');
        state.canRoll = false;
        centralDock.classList.add('rolling');
        hintDisplay.style.display = 'none'; // hide hint on roll
        
        setTimeout(() => {
            state.diceValue = Math.floor(Math.random() * 6) + 1;
            centralDock.classList.remove('rolling');
            diceValTxt.innerText = state.diceValue;
            
            instructionBanner.innerHTML = `Move ${state.diceValue} Steps`;
            state.canAnswer = true;
        }, 400);
    }

    function loadQuestion() {
        const q = words[state.currentQ];
        document.getElementById('q-word').innerText = q.word;
        document.getElementById('q-phonetic').innerText = q.phonetic;
        hintDisplay.style.display = 'none'; // Reset hint
        
        optionsContainer.innerHTML = '';
        const letters = ['A', 'B', 'C', 'D'];

        q.options.forEach((opt, idx) => {
            const btn = document.createElement('div');
            btn.className = 'opt-btn';
            btn.dataset.idx = idx;
            btn.innerHTML = `
                <div class="opt-letter">${letters[idx]}</div>
                <div class="opt-text">${opt}</div>
                <div class="opt-icon">🔊</div>
            `;
            btn.onclick = (e) => {
                // Determine if speaker was clicked to read Chinese, else answer question
                if(e.target.innerText === '🔊') {
                    e.stopPropagation();
                    playText(opt, 'zh-TW'); 
                } else {
                    checkAnswer(idx);
                }
            };
            optionsContainer.appendChild(btn);
        });
        
        feedbackBanner.classList.remove('show');
        instructionBanner.innerHTML = "Roll Dice";
        diceValTxt.innerText = "-";
        state.canRoll = true;
        state.canAnswer = false;
    }

    function checkAnswer(idx) {
        if(!state.canAnswer) {
            instructionBanner.innerText = "⚠️ Roll Dice First!";
            setTimeout(() => { if(state.canRoll) instructionBanner.innerHTML = "Roll Dice"; }, 1500);
            return;
        }

        state.canAnswer = false;
        const q = words[state.currentQ];
        const btns = document.querySelectorAll('.opt-btn');

        if(idx === q.correct) {
            btns[idx].classList.add('correct-ans');
            btns[idx].querySelector('.opt-icon').innerText = '✔';
            sfx('correct');
            
            const pointsEarned = state.diceValue * 5; 
            state.score += pointsEarned;
            
            feedbackIcon.innerText = "✔";
            feedbackText.innerHTML = `Correct! +${pointsEarned} points`;
            feedbackBanner.classList.remove('wrong-bg');
            feedbackBanner.classList.add('show');
            
            movePlayer(state.diceValue);
        } else {
            btns[idx].classList.add('wrong-ans');
            btns[q.correct].classList.add('correct-ans');
            btns[q.correct].querySelector('.opt-icon').innerText = '✔';
            
            sfx('wrong');
            state.energy = Math.max(0, state.energy - 1);
            
            feedbackIcon.innerText = "✖";
            feedbackText.innerHTML = `Try again!`;
            feedbackBanner.classList.add('wrong-bg');
            feedbackBanner.classList.add('show');
        }
        
        updateHUD();
    }

    function movePlayer(steps) {
        const pPawn = document.getElementById('player-pawn');
        pPawn.classList.add('jumping');
        setTimeout(() => pPawn.classList.remove('jumping'), 400);

        state.playerPos = Math.min(state.totalTiles, state.playerPos + steps);
        updatePawnPositions();
        updateHUD();

        if(state.playerPos >= state.totalTiles) {
            setTimeout(() => {
                showGameOver(true);
            }, 800);
        } else {
            setTimeout(moveNPC, 800);
        }
    }

    function moveNPC() {
        const npcSteps = Math.floor(Math.random() * 3) + 1;
        const notify = document.getElementById('npc-notification');
        notify.innerText = `${state.npcMascot} ${state.npcName} rolled a ${npcSteps}!`;
        notify.classList.add('show');
        setTimeout(() => notify.classList.remove('show'), 2500);

        const nPawn = document.getElementById('npc-pawn');
        nPawn.classList.add('jumping');
        setTimeout(() => nPawn.classList.remove('jumping'), 400);

        state.npcPos = Math.min(state.totalTiles, state.npcPos + npcSteps);
        updatePawnPositions();

        if(state.npcPos >= state.totalTiles) {
            setTimeout(() => {
                showGameOver(false);
            }, 800);
        }
    }

    function nextQuestion() {
        state.currentQ = (state.currentQ + 1) % words.length;
        loadQuestion();
    }

    function updateHUD() {
        document.getElementById('ui-energy-num').innerText = `${state.energy}/5`;
        document.getElementById('ui-score').innerText = state.score;
    }

    function resetGame(showSelectModal = false) {
        document.getElementById('game-over-modal').style.display = 'none';
        state.playerPos = 0; state.npcPos = 0; state.score = 25; state.energy = 3;
        state.currentQ = 0; state.canRoll = true; state.canAnswer = false;
        
        if (showSelectModal) {
            document.getElementById('char-select-modal').style.display = 'flex';
        } else {
            renderBoard(); loadQuestion(); updateHUD();
        }
    }
